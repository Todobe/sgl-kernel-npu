cmake_minimum_required(VERSION 3.20)
project(sgl_cann_ops LANGUAGES C CXX)
set(SGL_CANN_STAGE "vendor" CACHE STRING "Build stage: vendor or adapter")
add_subdirectory(examples/helloworld)
if(SGL_CANN_STAGE STREQUAL "adapter")
    include(cmake/torch_adapter.cmake)
    return()
elseif(NOT SGL_CANN_STAGE STREQUAL "vendor")
    message(FATAL_ERROR "Unknown SGL_CANN_STAGE: ${SGL_CANN_STAGE}")
endif()
find_package(ASC REQUIRED)
set(package_name sgl_kernel_npu)
set(vendor_name sgl_kernel_npu CACHE STRING "Custom OPP vendor" FORCE)
npu_op_package(${package_name} TYPE RUN
    CONFIG INSTALL_PATH ${CMAKE_BINARY_DIR})
include(cmake/cann_package.cmake)
add_subdirectory(examples/helloworld/op_kernel)
